# wallreco vision vocabulary
#
# What CLIP is asked to look for. Copy this file to
# ~/.config/wallreco/labels.txt and edit it — wallreco picks that copy up
# automatically and re-embeds the prompts the first time it notices a change.
#
# Format
# ------
#   [category] mode=one|many threshold=0.30 max=2 requires=other
#   tag = prompt | another prompt for the same tag
#   ~   = a prompt that is never emitted as a tag
#
# Every category is scored on its own: wallreco softmaxes the prompts inside
# one category and keeps whatever clears `threshold`. That is what makes the
# tags precise. Asking "is this forest, city, beach or desert?" gives a sharp
# answer; asking "which of these ninety things is it?" does not.
#
#   mode=one    at most one tag from this category (they are alternatives)
#   mode=many   several may fire, best first, up to `max`
#   threshold   minimum probability within the category, 0..1
#   requires    only ask this category if the named one above it answered
#   ~ lines     decoys. They soak up the probability of images that match
#               nothing in the category, so an abstract gradient does not get
#               called a forest just because forest was the closest of four
#               bad options. Every category needs several.
#
# Prompts are full English sentences on purpose — CLIP was trained on captions,
# so "a photograph of a cat" scores far better than "cat".
#
# If a tag fires where it should not, the fix is usually a better decoy rather
# than a higher threshold: a threshold turns the category off for everything,
# a decoy only takes the images that genuinely belong to it.


[style] mode=one threshold=0.34
anime    = anime artwork | a digital illustration in japanese anime style | an anime screenshot
pixelart = pixel art where the individual square pixels are visible as blocks | a retro video game sprite made of chunky pixels
painting = an oil painting on canvas | a traditional painted artwork | a watercolour painting
photo    = a photograph taken with a camera | a real photo of a real place
3d       = a 3d rendered scene | a computer generated cgi render | a blender render
lineart  = a minimal line drawing | black and white line art with no shading
vector   = a flat digital illustration with bold smooth shapes and clean edges | a stylised drawing with large areas of flat colour
~        = a blurry out of focus image
~        = a plain background with nothing on it
~        = a photograph of a screen or a user interface
~        = a smooth digital painting with soft shading

~         = an animal with no people nearby
